% Self-measuring skills row: guarantees a category line never wraps, instead of trusting a
% char-count estimate on the Java side. Measures its own rendered width with \sbox and
% steps the font down (small -> footnotesize -> scriptsize) until it fits \linewidth, so
% the check is against the exact glyph metrics XeTeX will actually use for this run.
\newcommand{\skillrowtext}[2]{\textbf{#1}{: #2}}
\newlength{\skillrowbox}
\newcommand{\skillrow}[2]{%
  \setbox0=\hbox{\small\skillrowtext{#1}{#2}}%
  \skillrowbox=\wd0
  \ifdim\skillrowbox>\linewidth
    \setbox0=\hbox{\footnotesize\skillrowtext{#1}{#2}}%
    \skillrowbox=\wd0
    \ifdim\skillrowbox>\linewidth
      \scriptsize\skillrowtext{#1}{#2}\\
    \else
      \footnotesize\skillrowtext{#1}{#2}\\
    \fi
  \else
    \small\skillrowtext{#1}{#2}\\
  \fi
}
